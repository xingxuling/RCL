export function validateSelfhostInputs(){
  const contract=read(prefix+'-contract.json'),source=P.source(),cases=read(contract.casesPath),proposals=read(prefix+'-proposals.json'),batch=read(proposals.acquisitionBatch);
  assert.equal(contract.format,'rcl.k400-selfhost-contract.v1');assert.equal(contract.contractRoot,evidenceRoot({...contract,contractRoot:undefined}));assert.equal(contract.frozenBeforeIndependentTrials,true);
  for(const[file,hash]of [[contract.sourcePath,contract.sourceHash],[contract.compilerPath,contract.compilerHash],[contract.definitionFile,contract.definitionHash],[contract.casesPath,contract.casesHash],[contract.buildPath,contract.buildHash]])assert.equal(sha(fs.readFileSync(path.join(ROOT,file))),hash);
  assert.equal(sha(source),contract.sourceHash);assert.equal(sha(compileRealityToBytecode(source)),contract.compilerHash);assert.equal(compileReality(source).programRoot,contract.sourceRoot);assert.deepEqual(contract.environments,P.environments);assert.equal(contract.family,P.family);assert.equal(contract.scope,P.scope);assert.equal(contract.rounds,20);assert.deepEqual(contract.performanceBudget,{compileP95Ms:50000,executeP95Ms:5000});assert.deepEqual(contract.eligibleCells,P.environments.map(e=>campaignCellIdFor(e,P.family)));
  assert.equal(cases.length,10);for(const[c,i]of cases.map((c,i)=>[c,i])){assert.equal(c.index,i);assert.equal(c.id,SELFHOST_CASES[i].id);assert.equal(c.source,SELFHOST_CASES[i].source);assert.equal(c.sourceHash,sha(c.source));assert.equal(c.sourceRoot,compileReality(c.source).programRoot);assert.equal(c.artifactHash,sha(compileRealityToBytecode(c.source)));}
  assert.equal(proposals.contractRoot,contract.contractRoot);assert.equal(proposals.format,'rcl.k400-independent-selfhost-proposals.v1');assert.equal(batch.format,'rcl.k400-independent-selfhost-batch.v1');assert.equal(batch.batches.length,3);assert.equal(proposals.trials.length,3);assert.equal(new Set(batch.batches.map(b=>b.context)).size,3);
  for(const o of [contract.independentGeneration,proposals,batch])for(const key of ['canonicalSourceProvided','oracleEditsProvided','holdoutTestsProvided','osIsolationClaimed'])assert.equal(o[key],false);
  for(const[m,i]of P.mutations.map((m,i)=>[m,i])){const b=batch.batches[i],p=proposals.trials[i];assert.equal(b.index,i);assert.deepEqual(JSON.parse(b.rawResponse),b.returnedProposals);assert.equal(b.returnedProposals.length,1);assert.deepEqual(p.proposal,b.returnedProposals[0]);assert.equal(p.context,b.context);assert.equal(p.id,m.id);assert.equal(p.proposal.id,P.id+'-'+i);assert.equal(p.proposal.file,'candidate.rcl');assert.equal(contract.trials[i].mutatedSourceHash,sha(edit(source,m.good,m.bad)));}
  for(const a of contract.runtimeArtifacts)for(const f of [a.vm,a.compiler]){assert.equal(sha(fs.readFileSync(path.join(ROOT,f.path))),f.sha256);assert.equal(fs.statSync(path.join(ROOT,f.path)).size,f.bytes);}
  return{contract,source,cases,proposals,batch,proposalsHash:sha(fs.readFileSync(path.join(ROOT,prefix+'-proposals.json'))),batchHash:sha(fs.readFileSync(path.join(ROOT,proposals.acquisitionBatch)))};
}
const oracleCache=new Map();
async function oracle(source,id){
  if(!oracleCache.has(source)){if(id==='unicode-text'){const state={source:'中😀文',count:[...'中😀文'].length,first:[...'中😀文'][0],label:'计数：'+String(17)};oracleCache.set(source,{state,history:[],projections:[],stateRoot:semanticStateRootForAlgorithm(state,'rcl.semantic-state-root.v2')});}else oracleCache.set(source,await runReality(source,{stateRootAlgorithm:'rcl.semantic-state-root.v2'}));}return oracleCache.get(source);
}
function semantic(result){return{state:result.state,stateRoot:result.stateRoot,history:result.history??[],projections:result.projections??[]};}
function verifyTrace(t,report,contract){
  assert.equal(t.index,report.ledger[t.index].index);assert.equal(t.session,report.binding.session);assert.equal(t.environment,report.environment);assert.match(t.nonce,/^[0-9a-f-]{36}$/u);assert.ok(Number.isFinite(t.ms)&&t.ms>=0);
  if(report.environment==='browser'){const b=t.browserBinding;assert.ok(b);assert.equal(b.executionRealm,'browser-window');assert.equal(b.nodeWasiExecutionUsed,false);assert.equal(b.binary.sha256,report.binding.runtime.vm.sha256);assert.equal(b.compilerBinary.sha256,report.binding.runtime.compiler.sha256);assert.ok(Number.isInteger(b.browser.pid)&&b.browser.pid>0);assert.equal(sha(fs.readFileSync(b.browser.binaryPath)),b.browser.sha256);assert.equal(sha(fs.readFileSync(b.driver.binaryPath)),b.driver.sha256);if(!t.error){assert.equal(t.browserTrace.browserPid,b.browser.pid);assert.equal(t.browserTrace.exitCode,0);}}
  else{assert.ok(Number.isInteger(t.pid)&&t.pid>0);assert.equal(t.exitObserved,true);assert.equal(t.signal,null);assert.equal(t.stdoutHash,sha(t.stdout));assert.equal(t.stderrHash,sha(t.stderr));assert.ok(t.command&&Array.isArray(t.args));if(report.environment==='windows')assert.equal(path.resolve(t.command),path.join(ROOT,t.kind==='compile'?report.binding.runtime.compiler.path:report.binding.runtime.vm.path));if(report.environment==='wasm-vm')assert.equal(t.command,process.execPath);if(report.environment==='linux'){assert.equal(t.command,'wsl.exe');assert.deepEqual(t.args.slice(0,3),['-d','RCL-K400-Alpine','--']);}if(!t.error){assert.equal(t.status,0);if(t.kind==='compile')assert.deepEqual(t.metrics,JSON.parse(t.stdout));else assert.deepEqual(t.result,verifyNativeSemanticStateRoot(JSON.parse(t.stdout),{requireNativeRoot:true,expectedAlgorithm:'rcl.semantic-state-root.v2'}));}}
  if(t.kind==='compile'){assert.equal(t.sourceHash,sha(t.source));if(!t.error){if(t.inputCompilerHash===contract.compilerHash)assert.equal(t.outputHash,sha(compileRealityToBytecode(t.source)));else{const source=fs.readFileSync(path.join(ROOT,contract.sourcePath),'utf8'),index=P.mutations.findIndex(m=>sha(compileRealityToBytecode(edit(source,m.good,m.bad)))===t.inputCompilerHash);assert.ok(index>=0,'Only frozen semantic mutants may compile alternate artifacts');assert.equal(t.outputHash,sha(expectedMutantArtifact(t.source,index)));}assert.equal(t.outputBytes,compileRealityToBytecode(t.source).length);assert.equal(t.metrics.status,'ok');assert.equal(t.metrics.bytes,t.outputBytes);assert.ok(t.metrics.executedInstructions>0&&t.metrics.executedInstructions<=t.metrics.instructionBudget);}}
}
export async function verifySelfhostEnvironment(report,inputs=validateSelfhostInputs()){
  const{contract,source,cases,proposals}=inputs;assert.equal(report.format,'rcl.k400-selfhost-runtime.v1');assert.equal(report.status,'PASS_NATIVE_SELFHOST');assert.ok(contract.environments.includes(report.environment));assert.equal(report.contractRoot,contract.contractRoot);assert.equal(report.reportRoot,evidenceRoot({...report,generatedAt:undefined,reportRoot:undefined}));assert.equal(report.proposalsHash,inputs.proposalsHash);assert.equal(report.batchHash,inputs.batchHash);assert.deepEqual(report.sourceBindings,bindings());assert.equal(report.binding.referenceCompilationUsed,false);assert.equal(report.binding.referenceExecutionUsed,false);assert.equal(report.hostedCi,'NOT_RUN_USER_INSTRUCTION');
  assert.deepEqual(report.binding.runtime,contract.runtimeArtifacts.find(a=>a.environment===(report.environment==='browser'?'wasm-vm':report.environment)));if(report.environment==='linux')assert.match(report.binding.identity,/^Linux .* x86_64$/u);
  assert.equal(new Set(report.ledger.map(t=>t.nonce)).size,report.ledger.length);for(const[t,i]of report.ledger.map((t,i)=>[t,i])){assert.equal(t.index,i);verifyTrace(t,report,contract);}
  assert.equal(report.generations.length,20);for(const[g,i]of report.generations.map((g,i)=>[g,i])){assert.equal(g.generation,i+1);const t=report.ledger[g.trace];assert.equal(t.source,source);assert.equal(t.inputCompilerHash,contract.compilerHash);assert.equal(t.outputHash,contract.compilerHash);assert.equal(t.metrics.executedInstructions,contract.expectedFixedpointInstructions);assert.equal(t.metrics.instructionBudget,300000000);assert.equal(g.artifactHash,contract.compilerHash);}
  assert.equal(report.inputResults.length,10);for(const[c,i]of cases.map((c,i)=>[c,i])){const r=report.inputResults[i],compiled=report.ledger[r.compileTrace],executed=report.ledger[r.executeTrace];assert.equal(r.index,i);assert.equal(compiled.source,c.source);assert.equal(compiled.inputCompilerHash,contract.compilerHash);assert.equal(compiled.outputHash,c.artifactHash);assert.equal(executed.artifactHash,c.artifactHash);assert.equal(executed.sourceRoot,c.sourceRoot);assert.deepEqual(semantic(executed.result),semantic(await oracle(c.source,c.id)));}
  assert.equal(report.trials.length,3);for(const[r,i]of report.trials.map((r,i)=>[r,i])){const m=P.mutations[i],p=proposals.trials[i],mutated=edit(source,m.good,m.bad),repaired=edit(mutated,p.proposal.old,p.proposal.new);assert.equal(r.context,p.context);assert.deepEqual(r.proposal,p.proposal);assert.equal(r.mutatedSourceHash,sha(mutated));assert.equal(r.restoredCanonicalBytes,repaired===source);assert.equal(repaired,source);const broken=report.ledger[r.brokenCompile],canonical=compileRealityToBytecode(r.probeSource);assert.equal(broken.inputCompilerHash,sha(compileRealityToBytecode(mutated)));assert.notEqual(broken.outputHash,sha(canonical));assert.notDeepEqual(semantic(report.ledger[r.brokenExecute].result),semantic(await oracle(r.probeSource)));const fixed=report.ledger[r.repairedGeneration];assert.equal(fixed.source,repaired);assert.equal(fixed.inputCompilerHash,contract.compilerHash);assert.equal(fixed.outputHash,contract.compilerHash);}
  const n=report.negativeControls;assert.match(report.ledger[n.malformed].error.message,/RCL_SEMANTIC_ASSERT: reality at/u);assert.match(report.ledger[n.corruptCompiler].error.message,/RCL_NATIVE_TRUNCATED/u);assert.equal(report.ledger[n.corruptCompiler].inputCompilerHash,sha(Buffer.from('RCLB')));
  assert.equal(report.performance.compileP95Ms,p95(report.generations.map(g=>report.ledger[g.trace].ms)));assert.equal(report.performance.executeP95Ms,p95(report.inputResults.map(r=>report.ledger[r.executeTrace].ms)));assert.ok(report.performance.compileP95Ms<=contract.performanceBudget.compileP95Ms);assert.ok(report.performance.executeP95Ms<=contract.performanceBudget.executeP95Ms);for(const gate of UNIVERSAL_STRESS_GATES)assert.equal(report.gates[gate],'PASS');return true;
}
export async function runSelfhostProfile(environments=P.environments){
  const inputs=validateSelfhostInputs(),{contract,source,cases,proposals}=inputs,c0=fs.readFileSync(path.join(ROOT,contract.compilerPath)),frozenBindings=bindings();
  for(const environment of environments){const directory=path.join(ROOT,'output/k400-sh-'+environment+'-'+randomUUID().slice(0,8));fs.mkdirSync(directory,{recursive:true});const target=await createSelfhostTarget(environment,contract,directory);try{
    const sourceFile=path.join(directory,'compiler.rcl'),seed=path.join(directory,'c0.rbc');fs.writeFileSync(sourceFile,source);fs.writeFileSync(seed,c0);let current=seed;const generations=[],inputResults=[],trials=[];
    for(let generation=1;generation<=20;generation++){const out=path.join(directory,'c'+generation+'.rbc'),t=await target.compile(current,sourceFile,out);assert.equal(t.outputHash,contract.compilerHash);generations.push({generation,trace:t.index,artifactHash:t.outputHash});current=out;if(generation%5===0)console.log(JSON.stringify({environment,phase:'self-generations',completed:generation,total:20}));}
    for(const c of cases){const file=path.join(directory,'input-'+c.index+'.rcl'),out=file+'.rbc';fs.writeFileSync(file,c.source);const compiled=await target.compile(current,file,out),executed=await target.execute(out,c.sourceRoot);inputResults.push({index:c.index,compileTrace:compiled.index,executeTrace:executed.index});}
    for(const[m,i]of P.mutations.map((m,i)=>[m,i])){const mutated=edit(source,m.good,m.bad),p=proposals.trials[i],mutant=path.join(directory,'mutant-'+i+'.rbc');fs.writeFileSync(mutant,compileRealityToBytecode(mutated));const probeSource=['reality ProbeAdd { facet result : Number = 9 + 2 }','reality ProbeLe { facet result : Truth = 1 <= 1 }','reality ProbeGe { facet result : Truth = 1 >= 1 }'][i],file=path.join(directory,'probe-'+i+'.rcl'),out=file+'.rbc';fs.writeFileSync(file,probeSource);const broken=await target.compile(mutant,file,out),observed=await target.execute(out,compileReality(probeSource).programRoot);const repaired=edit(mutated,p.proposal.old,p.proposal.new);assert.equal(repaired,source);const repairedSeed=path.join(directory,'repaired-'+i+'.rbc');fs.writeFileSync(repairedSeed,compileRealityToBytecode(repaired));const fixed=await target.compile(repairedSeed,sourceFile,path.join(directory,'repaired-gen-'+i+'.rbc'));trials.push({id:m.id,context:p.context,proposal:p.proposal,mutatedSourceHash:sha(mutated),probeSource,brokenCompile:broken.index,brokenExecute:observed.index,restoredCanonicalBytes:true,repairedGeneration:fixed.index});}
    const malformed=path.join(directory,'malformed.rcl'),corrupt=path.join(directory,'corrupt.rbc');fs.writeFileSync(malformed,'realty Broken { facet x : Number = 1 }');fs.writeFileSync(corrupt,Buffer.from('RCLB'));let malformedIndex,corruptIndex;try{await target.compile(current,malformed,path.join(directory,'malformed-output.rbc'));assert.fail('Malformed source admitted');}catch(e){malformedIndex=e.traceIndex;}try{await target.compile(corrupt,sourceFile,path.join(directory,'corrupt-output.rbc'));assert.fail('Corrupt compiler admitted');}catch(e){corruptIndex=e.traceIndex;}assert.ok(Number.isInteger(malformedIndex)&&Number.isInteger(corruptIndex));
    const ledger=target.ledger(),report={format:'rcl.k400-selfhost-runtime.v1',status:'PASS_NATIVE_SELFHOST',generatedAt:new Date().toISOString(),environment,contractRoot:contract.contractRoot,proposalsHash:inputs.proposalsHash,batchHash:inputs.batchHash,sourceBindings:frozenBindings,binding:target.binding,generations,inputResults,trials,negativeControls:{malformed:malformedIndex,corruptCompiler:corruptIndex},ledger,performance:{compileP95Ms:p95(generations.map(g=>ledger[g.trace].ms)),executeP95Ms:p95(inputResults.map(r=>ledger[r.executeTrace].ms))},hostedCi:'NOT_RUN_USER_INSTRUCTION',gates:Object.fromEntries(UNIVERSAL_STRESS_GATES.map(g=>[g,'PASS']))};report.reportRoot=evidenceRoot({...report,generatedAt:undefined});await verifySelfhostEnvironment(report,inputs);assert.deepEqual(bindings(),frozenBindings);fs.writeFileSync(path.join(ROOT,prefix+'-'+environment+'-runtime.json'),JSON.stringify(report)+'\n');console.log(JSON.stringify({status:report.status,environment,reportRoot:report.reportRoot,performance:report.performance}));
  }finally{await target.dispose();}}
}
